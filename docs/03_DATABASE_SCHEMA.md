FilmedIn - Supabase Database SchemaRun this SQL in your Supabase SQL Editor to set up the database.-- 1. Profiles Table (Links to Supabase Auth)CREATE TABLE profiles (  id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,  username TEXT UNIQUE NOT NULL,  created_at TIMESTAMPTZ DEFAULT NOW());-- 2. Playlists TableCREATE TABLE playlists (  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE NOT NULL,  name TEXT NOT NULL,  type TEXT CHECK (type IN ('system', 'custom')) NOT NULL DEFAULT 'custom',  created_at TIMESTAMPTZ DEFAULT NOW());-- 3. Playlist Items Table (Junction table for Playlists and TMDB items)CREATE TABLE playlist_items (  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,  playlist_id UUID REFERENCES playlists(id) ON DELETE CASCADE NOT NULL,  tmdb_id INTEGER NOT NULL,  media_type TEXT CHECK (media_type IN ('movie', 'tv')) NOT NULL,  added_at TIMESTAMPTZ DEFAULT NOW(),  UNIQUE(playlist_id, tmdb_id, media_type));-- 4. Wallpapers Table (Caching mechanism)CREATE TABLE wallpapers (  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,  tmdb_id INTEGER NOT NULL,  media_type TEXT CHECK (media_type IN ('movie', 'tv')) NOT NULL,  desktop_url TEXT,  mobile_url TEXT,  prompt_used TEXT,  created_at TIMESTAMPTZ DEFAULT NOW(),  UNIQUE(tmdb_id, media_type));-- Enable Row Level Security (RLS)ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;ALTER TABLE playlists ENABLE ROW LEVEL SECURITY;ALTER TABLE playlist_items ENABLE ROW LEVEL SECURITY;ALTER TABLE wallpapers ENABLE ROW LEVEL SECURITY;-- RLS Policies-- Profiles: Users can only view/edit their own profileCREATE POLICY "Public profiles are viewable by everyone." ON profiles FOR SELECT USING (true);CREATE POLICY "Users can insert their own profile." ON profiles FOR INSERT WITH CHECK (auth.uid() = id);CREATE POLICY "Users can update own profile." ON profiles FOR UPDATE USING (auth.uid() = id);-- Playlists &amp; Items: Users can only manage their ownCREATE POLICY "Users can view their own playlists." ON playlists FOR SELECT USING (auth.uid() = user_id);CREATE POLICY "Users can insert their own playlists." ON playlists FOR INSERT WITH CHECK (auth.uid() = user_id);CREATE POLICY "Users can delete their own playlists." ON playlists FOR DELETE USING (auth.uid() = user_id);CREATE POLICY "Users can view their own playlist items." ON playlist_items FOR SELECT USING (auth.uid() IN (SELECT user_id FROM playlists WHERE id = playlist_id));CREATE POLICY "Users can insert their own playlist items." ON playlist_items FOR INSERT WITH CHECK (auth.uid() IN (SELECT user_id FROM playlists WHERE id = playlist_id));CREATE POLICY "Users can delete their own playlist items." ON playlist_items FOR DELETE USING (auth.uid() IN (SELECT user_id FROM playlists WHERE id = playlist_id));-- Wallpapers: Publicly viewable (cached), but only auth users can generate/addCREATE POLICY "Wallpapers are viewable by everyone." ON wallpapers FOR SELECT USING (true);CREATE POLICY "Auth users can insert wallpapers." ON wallpapers FOR INSERT WITH CHECK (auth.role() = 'authenticated');Default Playlists LogicWhen a new user signs up, a database trigger (or a Next.js Server Action immediately after signup) will automatically create three 'system' type playlists:"Watching""Plan to Watch""Watched"
